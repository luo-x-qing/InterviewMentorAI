1. RAG服务实现

创建backend_python/app/services/rag_service.py：

# rag_service.py
from typing import List, Dict, Any
from langchain_community.vectorstores import Chroma
from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_community.retrievers import BM25Retriever
from langchain.retrievers import EnsembleRetriever
from langchain.text_splitter import RecursiveCharacterTextSplitter
import os

class RAGService:
    def __init__(self):
        # 初始化嵌入模型
        self.embeddings = HuggingFaceEmbeddings(
            model_name="BAAI/bge-large-zh-v1.5"
        )
        
        # 初始化向量数据库
        self.vector_store_path = os.path.join(
            os.path.dirname(__file__), 
            "../knowledge_base/vector_store"
        )
        self.vector_store = Chroma(
            persist_directory=self.vector_store_path,
            embedding_function=self.embeddings
        )
        
        # 初始化文本分割器
        self.text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=500,
            chunk_overlap=50,
            length_function=len,
        )
        
        # 初始化混合检索器
        self._setup_retrievers()
    
    def _setup_retrievers(self):
        """设置混合检索器"""
        # 向量检索器
        vector_retriever = self.vector_store.as_retriever(
            search_kwargs={"k": 5}
        )
        
        # BM25检索器（需要先加载文档）
        self._load_bm25_index()
        
        # 混合检索器
        self.ensemble_retriever = EnsembleRetriever(
            retrievers=[self.bm25_retriever, vector_retriever],
            weights=[0.3, 0.7]
        )
    
    def _load_bm25_index(self):
        """加载BM25索引"""
        # 从知识库加载文档
        documents = self._load_documents()
        
        # 创建BM25检索器
        self.bm25_retriever = BM25Retriever.from_documents(documents)
        self.bm25_retriever.k = 5
    
    def _load_documents(self) -> List[str]:
        """加载知识库文档"""
        documents = []
        knowledge_base_path = os.path.join(
            os.path.dirname(__file__), 
            "../knowledge_base"
        )
        
        # 遍历知识库目录
        for root, dirs, files in os.walk(knowledge_base_path):
            for file in files:
                if file.endswith(".txt") or file.endswith(".md"):
                    file_path = os.path.join(root, file)
                    with open(file_path, 'r', encoding='utf-8') as f:
                        documents.append(f.read())
        
        return documents
    
    def add_documents(self, documents: List[str], metadatas: List[Dict] = None):
        """添加文档到知识库"""
        # 分割文档
        splits = self.text_splitter.split_documents(documents)
        
        # 添加到向量数据库
        self.vector_store.add_documents(splits, metadatas=metadatas)
        
        # 重新加载BM25索引
        self._load_bm25_index()
    
    def retrieve(self, query: str, k: int = 5) -> List[Dict[str, Any]]:
        """检索相关文档"""
        # 使用混合检索
        results = self.ensemble_retriever.get_relevant_documents(query)
        
        # 格式化结果
        formatted_results = []
        for doc in results:
            formatted_results.append({
                "content": doc.page_content,
                "metadata": doc.metadata,
                "score": getattr(doc, "score", None)
            })
        
        return formatted_results
    
    def context_aware_retrieval(self, query: str, dialogue_context: str) -> List[Dict[str, Any]]:
        """上下文感知检索"""
        # 分析对话上下文，生成增强查询
        enhanced_query = self._enhance_query_with_context(query, dialogue_context)
        
        # 执行检索
        return self.retrieve(enhanced_query)
    
    def _enhance_query_with_context(self, query: str, context: str) -> str:
        """根据上下文增强查询"""
        # 简单实现：将查询和上下文组合
        # 实际项目中可以使用LLM进行查询改写
        return f"{query} {context}"

2. RAG API实现

创建backend_python/app/api/rag.py：

# rag.py
from fastapi import APIRouter, HTTPException
from typing import List, Dict, Any
from ..services.rag_service import RAGService
from ..models.schemas import RetrievalRequest, RetrievalResponse

router = APIRouter()
rag_service = RAGService()

@router.post("/retrieve", response_model=RetrievalResponse)
async def retrieve_documents(request: RetrievalRequest):
    """检索相关文档"""
    try:
        results = rag_service.retrieve(
            query=request.query,
            k=request.k
        )
        return RetrievalResponse(
            success=True,
            results=results
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/context-aware-retrieve", response_model=RetrievalResponse)
async def context_aware_retrieve(request: RetrievalRequest):
    """上下文感知检索"""
    try:
        results = rag_service.context_aware_retrieval(
            query=request.query,
            dialogue_context=request.context
        )
        return RetrievalResponse(
            success=True,
            results=results
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@router.post("/add-documents")
async def add_documents(documents: List[Dict[str, Any]]):
    """添加文档到知识库"""
    try:
        rag_service.add_documents(documents)
        return {"success": True, "message": "文档添加成功"}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

class LLMService:
    def __init__(self):
        # 现有初始化代码...
        self.rag_service = RAGService()
    
    def rag_enhanced_evaluation(self, question: str, answer: str, context: str) -> str:
        """RAG增强的评估"""
        # 1. 检索相关知识
        relevant_knowledge = self.rag_service.context_aware_retrieval(
            f"评估问题：{question}，回答：{answer}",

        knowledge_text = "\n".join([k["content"] for k in relevant_knowledge])

        你是一个专业的面试评估专家。请基于以下信息评估应聘者的回答：
        
        面试问题：{question}

        相关知识参考：
        {knowledge_text}

        1. 技术准确性
        2. 回答完整性
        3. 表达清晰度
        4. 实际应用能力
        
        评估结果：
        """
        
        # 3. 调用LLM进行评估
        return self.generate_report(enhanced_prompt)

# Python基础面试题

## 问题1：Python中的列表和元组有什么区别？
**标准答案：**
- 列表（list）是可变的，可以修改元素
- 元组（tuple）是不可变的，创建后不能修改
- 列表使用方括号[]，元组使用圆括号()
- 元组可以作为字典的键，列表不能

**评估要点：**
- 是否理解可变性和不可变性的概念
- 是否知道实际应用场景
- 是否能举例说明

## 问题2：解释Python中的装饰器
**标准答案：**
- 装饰器是一种设计模式，用于在不修改原函数的情况下增加功能
- 使用@语法糖
- 本质上是一个接受函数作为参数的高阶函数
- 常用于日志记录、权限检查、性能计时等

**评估要点：**
- 是否理解装饰器的基本原理
- 是否能编写简单的装饰器
- 是否知道实际应用场景

2. 监控配置

添加监控和日志记录：

# 在RAG服务中添加监控
import logging
from datetime import datetime

logger = logging.getLogger(__name__)

class MonitoredRAGService(RAGService):
    def retrieve(self, query: str, k: int = 5):
        """带监控的检索"""
        start_time = datetime.now()
        
        try:
            results = super().retrieve(query, k)
            
            # 记录成功日志
            duration = (datetime.now() - start_time).total_seconds()
            logger.info(f"检索成功: query='{query}', k={k}, duration={duration:.3f}s, results={len(results)}")
            
            return results
            
        except Exception as e:
            # 记录错误日志
            duration = (datetime.now() - start_time).total_seconds()
            logger.error(f"检索失败: query='{query}', k={k}, duration={duration:.3f}s, error={str(e)}")
            raise
